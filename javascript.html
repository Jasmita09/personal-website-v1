// Security Tips Array var securityTips = [ "Use strong, unique passwords for every account.", "Enable Multi-Factor Authentication (MFA) on all primary accounts.", "Be cautious of suspicious links or unknown email attachments.", "Keep your software and operating systems regularly updated.", "Always lock your workstation screen when stepping away." ]; var currentTipIndex = 0; function displayNextTip() { var tipDisplay = document.getElementById("tip-display"); if (tipDisplay) { tipDisplay.innerHTML = securityTips[currentTipIndex]; currentTipIndex = (currentTipIndex + 1) % securityTips.length; } } // Quiz Scoring var analyticsCount = 0; var engineeringCount = 0; var selectedAnswers = 0; function selectAnalytics() { analyticsCount += 1; selectedAnswers += 1; evaluateQuiz(); } function selectEngineering() { engineeringCount += 1; selectedAnswers += 1; evaluateQuiz(); } function evaluateQuiz() { var resultElement = document.getElementById("quiz-result"); if (resultElement && selectedAnswers === 3) { if (analyticsCount >= 2) { resultElement.innerHTML = "Result: You align with Data Analytics & Predictive Modeling!"; } else { resultElement.innerHTML = "Result: You align with Data Engineering & Infrastructure!"; } } } function resetQuizState() { analyticsCount = 0; engineeringCount = 0; selectedAnswers = 0; var resultElement = document.getElementById("quiz-result"); if (resultElement) { resultElement.innerHTML = "Your result will appear here..."; } } // Initialize Page Listeners document.addEventListener("DOMContentLoaded", function() { var tipButton = document.getElementById("tip-btn"); if (tipButton) { tipButton.addEventListener("click", displayNextTip); } var btn1a = document.getElementById("btn-option1a"); var btn1b = document.getElementById("btn-option1b"); var btn2a = document.getElementById("btn-option2a"); var btn2b = document.getElementById("btn-option2b"); var btn3a = document.getElementById("btn-option3a"); var btn3b = document.getElementById("btn-option3b"); var resetBtn = document.getElementById("btn-reset"); if (btn1a) { btn1a.addEventListener("click", selectAnalytics); btn1b.addEventListener("click", selectEngineering); btn2a.addEventListener("click", selectAnalytics); btn2b.addEventListener("click", selectEngineering); btn3a.addEventListener("click", selectAnalytics); btn3b.addEventListener("click", selectEngineering); } if (resetBtn) { resetBtn.addEventListener("click", resetQuizState); } });
